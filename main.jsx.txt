function hashHue(str) {
  let h = 0;
  for (let i = 0; i < str.length; i++) h = str.charCodeAt(i) + ((h << 5) - h);
  return Math.abs(h) % 360;
}

function posterGradient(seed) {
  const hue = hashHue(seed);
  const hue2 = (hue + 38) % 360;
  const hue3 = (hue2 + 22) % 360;
  return `linear-gradient(155deg, hsl(${hue} 68% 22%) 0%, hsl(${hue2} 70% 11%) 62%, hsl(${hue3} 55% 6%) 100%)`;
}

function dominantVerdict(split) {
  return VERDICTS.reduce((best, v) => (split[v.key] > split[best.key] ? v : best), VERDICTS[0]);
}

function computeSplit(movie, userVerdict) {
  const counts = {};
  VERDICTS.forEach((v) => { counts[v.key] = Math.round((movie.votes * movie.split[v.key]) / 100); });
  if (userVerdict) counts[userVerdict] += 1;
  const total = Object.values(counts).reduce((a, b) => a + b, 0) || 1;
  const pct = {};
  VERDICTS.forEach((v) => { pct[v.key] = Math.round((counts[v.key] / total) * 1000) / 10; });
  return pct;
}

function TypeIcon({ type, size = 16 }) {
  if (type === "show") return <Tv size={size} strokeWidth={1.5} />;
  if (type === "anime") return <Sparkles size={size} strokeWidth={1.5} />;
  return <Film size={size} strokeWidth={1.5} />;
}

function MeterBar({ split }) {
  const label = VERDICTS.map((v) => `${v.label} ${split[v.key]} percent`).join(", ");
  return (
    <div className="meter-bar" role="img" aria-label={`Community verdict: ${label}`}>
      {VERDICTS.map((v) => (
        <span key={v.key} className="meter-seg" style={{ width: `${split[v.key]}%`, background: v.color }} />
      ))}
    </div>
  );
}

function MeterLegend({ split }) {
  return (
    <div className="meter-legend">
      {VERDICTS.map((v) => (
        <div key={v.key} className="meter-legend-item">
          <span className="meter-legend-emoji">{v.emoji}</span>

          <span className="meter-legend-pct" style={{ color: v.color }}>{split[v.key]}%</span>
        </div>
      ))}
    </div>
  );
}

function PosterCard({ movie, inWatchlist, onToggleWatchlist, onOpen, userVerdict }) {
  const dom = dominantVerdict(movie.split);
  const verdictInfo = userVerdict ? VERDICTS.find((v) => v.key === userVerdict) : null;
  return (
    <div className="poster-card">
      <button
        className="poster"
        style={{ backgroundImage: posterGradient(movie.id) }}
        onClick={onOpen}
        aria-label={`Open details for ${movie.title}`}
      >
        <span className="poster-watermark" aria-hidden="true"><TypeIcon type={movie.type} size={64} /></span>
        <span className="poster-scrim" aria-hidden="true" />
        <span className="poster-info">
          <span className="poster-title">{movie.title}</span>
          <span className="poster-meta-row">
            <span className="lean-dot" style={{ background: dom.color }} aria-hidden="true" />
            <span className="poster-meta">{movie.year} · {movie.meta}</span>
          </span>
        </span>
        {movie.status === "upcoming" && <span className="poster-badge">Coming Soon</span>}
      </button>
      <button
        className={`bookmark-btn ${inWatchlist ? "active" : ""}`}
        onClick={(e) => { e.stopPropagation(); onToggleWatchlist(); }}
        aria-pressed={inWatchlist}
        aria-label={inWatchlist ? `Remove ${movie.title} from watchlist` : `Add ${movie.title} to watchlist`}
      >
        <Bookmark size={15} fill={inWatchlist ? "currentColor" : "none"} />
      </button>
      {verdictInfo && <span className="poster-uservote" title="Your verdict">{verdictInfo.emoji}</span>}
    </div>
  );
}

function Rail({ title, icon, movies, watchlist, toggleWatchlist, openMovie, userVerdicts }) {
  const trackRef = useRef(null);
  if (!movies.length) return null;
  const scroll = (dir) => {
    if (!trackRef.current) return;
    trackRef.current.scrollBy({ left: dir * trackRef.current.clientWidth * 0.85, behavior: "smooth" });
  };
  return (
    <section className="rail">
      <div className="rail-head">
        <h2 className="rail-title">{icon} {title}</h2>
        <div className="rail-arrows">
          <button aria-label={`Scroll ${title} left`} onClick={() => scroll(-1)}><ChevronLeft size={18} /></button>
          <button aria-label={`Scroll ${title} right`} onClick={() => scroll(1)}><ChevronRight size={18} /></button>
        </div>
      </div>
      <div className="rail-track" ref={trackRef}>
        {movies.map((m) => (
          <PosterCard
            key={m.id}
            movie={m}
            inWatchlist={watchlist.has(m.id)}
            onToggleWatchlist={() => toggleWatchlist(m.id)}
            onOpen={() => openMovie(m.id)}
            userVerdict={userVerdicts[m.id]}
          />
        ))}
      </div>
    </section>
  );
}

/* ============================== APP ============================== */

export default function App() {
  const [view, setView] = useState("home");
  const [watchlist, setWatchlist] = useState(() => new Set());
  const [userVerdicts, setUserVerdicts] = useState({});
  const [ratedOrder, setRatedOrder] = useState([]);
  const [searchQuery, setSearchQuery] = useState("");
  const [typeFilter, setTypeFilter] = useState("all");
  const [genreFilters, setGenreFilters] = useState(() => new Set());
  const [sortBy, setSortBy] = useState("trending");
  const [selectedId, setSelectedId] = useState(null);
  const [authOpen, setAuthOpen] = useState(false);
  const [recapOpen, setRecapOpen] = useState(false);
  const [user, setUser] = useState(null);

  const toggleWatchlist = (id) => {
    setWatchlist((prev) => {
      const next = new Set(prev);
      if (next.has(id)) next.delete(id); else next.add(id);
      return next;
    });
  };

  const rate = (id, verdict) => {
    setUserVerdicts((prev) => ({ ...prev, [id]: verdict }));
    setRatedOrder((prev) => (prev.includes(id) ? [...prev.filter((x) => x !== id), id] : [...prev, id]));
  };

  const openMovie = (id) => setSelectedId(id);
  const closeMovie = () => setSelectedId(null);
  const signIn = (name) => { setUser({ name: name && name.trim() ? name.trim() : "Guest Cinephile" }); setAuthOpen(false); };

  const handleSearchChange = (v) => {
    setSearchQuery(v);
    if (v) setView("discover");
  };

  const toggleGenre = (g) => {
    setGenreFilters((prev) => {
      const next = new Set(prev);
      if (next.has(g)) next.delete(g); else next.add(g);
      return next;
    });
  };

  const filteredTitles = useMemo(() => {
    let list = TITLES.filter((m) => {
      if (typeFilter !== "all" && m.type !== typeFilter) return false;
      if (genreFilters.size > 0 && !m.genres.some((g) => genreFilters.has(g))) return false;
      if (searchQuery && !m.title.toLowerCase().includes(searchQuery.toLowerCase())) return false;
      return true;
    });
    if (sortBy === "top") list = [...list].sort((a, b) => b.split.perfection - a.split.perfection);
    else if (sortBy === "az") list = [...list].sort((a, b) => a.title.localeCompare(b.title));
    else if (sortBy === "newest") list = [...list].sort((a, b) => b.year - a.year);
    else list = [...list].sort((a, b) => b.votes - a.votes);
    return list;
  }, [typeFilter, genreFilters, searchQuery, sortBy]);

  const trending = useMemo(() => [...TITLES].filter((m) => m.status === "released").sort((a, b) => b.votes - a.votes).slice(0, 10), []);
  const popular = useMemo(() => [...TITLES].filter((m) => m.status === "released").sort((a, b) => b.split.perfection - a.split.perfection).slice(0, 10), []);
  const upcoming = useMemo(() => TITLES.filter((m) => m.status === "upcoming"), []);
  const spotlight = trending[0];

  const selectedMovie = selectedId ? TITLES.find((m) => m.id === selectedId) : null;
  const similar = useMemo(() => {
    if (!selectedMovie) return [];
    return TITLES.filter((m) => m.id !== selectedMovie.id && m.genres.some((g) => selectedMovie.genres.includes(g))).slice(0, 6);
  }, [selectedMovie]);

  const watchlistTitles = useMemo(() => TITLES.filter((m) => watchlist.has(m.id)), [watchlist]);
  const ratedTitles = useMemo(() => ratedOrder.slice().reverse().map((id) => TITLES.find((m) => m.id === id)).filter(Boolean), [ratedOrder]);

  const level = useMemo(() => {
    const n = ratedOrder.length;
    if (n === 0) return "New Recruit";
    if (n < 5) return "Casual Viewer";
    if (n < 10) return "Cinephile in Training";
    if (n < 15) return "Cinephile";
    return "Certified Culture Head";
  }, [ratedOrder]);

  const recapStats = useMemo(() => {
    const counts = { perfection: 0, goforit: 0, timepass: 0, skipit: 0 };
    Object.values(userVerdicts).forEach((v) => { counts[v] = (counts[v] || 0) + 1; });
    const favEntry = Object.entries(counts).sort((a, b) => b[1] - a[1])[0];
    const genreCount = {};
    [...watchlistTitles, ...ratedTitles].forEach((m) => m.genres.forEach((g) => { genreCount[g] = (genreCount[g] || 0) + 1; }));
    const genreEntry = Object.entries(genreCount).sort((a, b) => b[1] - a[1])[0];
    return {
      rated: ratedOrder.length,
      watchlist: watchlist.size,
      favoriteVerdict: favEntry && favEntry[1] > 0 ? favEntry[0] : null,
      favoriteGenre: genreEntry ? genreEntry[0] : null,
    };
  }, [userVerdicts, watchlistTitles, ratedTitles, ratedOrder, watchlist]);

  return (
    <>
      <style>{STYLES}</style>
      <div className="moctale-root">
        <Navbar
          view={view}
          setView={setView}
          watchlistCount={watchlist.size}
          searchQuery={searchQuery}
          onSearchChange={handleSearchChange}
          user={user}
          onOpenAuth={() => setAuthOpen(true)}
        />

        <main className="main-content">
          {view === "home" && (
            <HomeView
              spotlight={spotlight}
              trending={trending}
              popular={popular}
              upcoming={upcoming}
              watchlist={watchlist}
              toggleWatchlist={toggleWatchlist}
              openMovie={openMovie}
              userVerdicts={userVerdicts}
              onOpenRecap={() => setRecapOpen(true)}
            />
          )}
          {view === "discover" && (
            <DiscoverView
              titles={filteredTitles}
              searchQuery={searchQuery}
              onSearchChange={handleSearchChange}
              typeFilter={typeFilter}
              setTypeFilter={setTypeFilter}
              genreFilters={genreFilters}
              toggleGenre={toggleGenre}
              sortBy={sortBy}
              setSortBy={setSortBy}
              watchlist={watchlist}
              toggleWatchlist={toggleWatchlist}
              openMovie={openMovie}
              userVerdicts={userVerdicts}
            />
          )}
          {view === "watchlist" && (
            <WatchlistView
              titles={watchlistTitles}
              watchlist={watchlist}
              toggleWatchlist={toggleWatchlist}
              openMovie={openMovie}
              userVerdicts={userVerdicts}
              goDiscover={() => setView("discover")}
            />
          )}
          {view === "profile" && (
            <ProfileView
              user={user}
              level={level}
              ratedCount={ratedOrder.length}
              watchlistCount={watchlist.size}
              ratedTitles={ratedTitles}
              userVerdicts={userVerdicts}
              onOpenAuth={() => setAuthOpen(true)}
              onOpenRecap={() => setRecapOpen(true)}
              openMovie={openMovie}
            />
          )}
        </main>

        <MobileTabBar view={view} setView={setView} watchlistCount={watchlist.size} />

        {selectedMovie && (
          <MovieDetail
            movie={selectedMovie}
            onClose={closeMovie}
            inWatchlist={watchlist.has(selectedMovie.id)}
            onToggleWatchlist={() => toggleWatchlist(selectedMovie.id)}
            userVerdict={userVerdicts[selectedMovie.id]}
            onRate={(v) => rate(selectedMovie.id, v)}
            similar={similar}
            onOpenSimilar={(id) => setSelectedId(id)}
          />
        )}

        {authOpen && <AuthModal onClose={() => setAuthOpen(false)} onSignIn={signIn} />}
        {recapOpen && (
          <RecapModal
            stats={recapStats}
            level={level}
            onClose={() => setRecapOpen(false)}
            onExplore={() => { setRecapOpen(false); setView("discover"); }}
          />
        )}

        <footer className="site-footer">
          <p>Fan-made concept UI inspired by moctale.in — built as a design demo, not affiliated with Moctale or Men of Culture.</p>
        </footer>
      </div>
    </>
  );
}
